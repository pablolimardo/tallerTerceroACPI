# CON función — escrita una sola vez, modular y limpia (bien)
def evaluar_zona(nombre, temp, hum):
    """Evalúa las condiciones ambientales de una zona concreta."""
    if temp > 30.0:
        estado = "🔴 ALERTA_CALOR"
    elif hum > 80.0:
        estado = "🔵 ALERTA_HUMEDAD"
    elif temp >= 18.0 and temp <= 26.0:
        estado = "🟢 CONFORT"
    else:
        estado = "🟡 FUERA_RANGO"
    
    return f"[{nombre.upper()}] Estado: {estado} (T: {temp}°C | H: {hum}%)"

# Llamadas reutilizables
res1 = evaluar_zona("Living", 22.5, 50)
res2 = evaluar_zona("Cocina", 31.0, 60)
res3 = evaluar_zona("Dormitorio", 18.3, 45)
res4 = evaluar_zona("Baño", 24.1, 85)

class Habitacion:
    def __init__(self, nombre, temp, hum):
        self.nombre = nombre       # Atributo público
        self.temp = temp           # Atributo público
        self.hum = hum             # Atributo público
        self.ventilador = False   # Estado inicial

    def estado(self):
        """Método que determina la condición de confort."""
        if self.temp > 26.0:
            return "CALOR"
        return "CONFORT"

    def regular(self):
        """Método que actúa sobre el ventilador."""
        if self.temp > 26.0:
            self.ventilador = True
        else:
            self.ventilador = False
        return self.ventilador

# Crear instancias independientes
living = Habitacion("Living", 22.0, 55)
cocina = Habitacion("Cocina", 28.5, 70)

cocina.regular()  # Enciende el ventilador de la cocina únicamente

class Sensor:
    def __init__(self, pin, zona):

        self.zona = zona

        pass  # Método genérico

        return {"temp": 24.2, "hum": 62}

    def leer(self):
        return True  # Detectó presencia

# Polimorfismo: Bucle uniforme sobre sensores distintos
lista_sensores = [SensorDHT22(4, "Living"), SensorMovimiento(12, "Hall")]
for s in lista_sensores:

Estructura `setup()` y `loop()` en el Microcontrolador

// Código del microcontrolador ESP32 en C++
#include <DHT.h>

#define DHTPIN 4
#define RELE_PIN 16
#define DHTTYPE DHT22

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(115200);
  pinMode(RELE_PIN, OUTPUT);
  dht.begin();
}

void loop() {
  float temp = dht.readTemperature();
  
  if (temp > 26.0) {
    digitalWrite(RELE_PIN, HIGH); // Enciende ventilador
  } else {
    digitalWrite(RELE_PIN, LOW);  // Apaga ventilador
  }
  delay(2000);
}
